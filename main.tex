\begin{frame}[fragile]{Краткая история тестирования в Perl}
    \begin{itemize}
        \item Для perl был разработан TAP (\textit{Test Anything Protocol})
        \vspace{0.25em}
        \item Разработчики вручную писали строки вывода, например: \texttt{print "ok - Test passed\textbackslash n";}
        \vspace{0.25em}
        \item Появились подпрограммы для автоматизации: \texttt{ok()} и подобные
        \vspace{0.25em}
        \item Стали появляться библиотеки объединиящие такие функции
        \vspace{0.25em}
        \item Возникали проблемы из-за их несовместимости
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Краткая история тестирования в Perl}
    \begin{itemize}
        \item Michael Schwern выпустил \texttt{Test::Builder} как совместимую основу, породив экосистему \texttt{Test::Builder}
        \vspace{0.25em}
        \item Schwern и другие пытались переработать \texttt{Test::Builder} для исправления недостатков, но проект закрыли
        \vspace{0.25em}
        \item Chad Granum выпустил \texttt{Test2} и обновил \texttt{Test::Builder}, превратив его в совместимый слой поверх \texttt{Test2}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Test::More - заслуженный ветеран}
    \begin{itemize}\setlength\itemsep{1.2ex}
        \item Появился в 2001 году
        \item Стандарт де-факто для Perl-тестирования
        \vspace{0.25em}
        \begin{itemize}
            \setlength{\itemsep}{0.5ex}
            \item Входит в ядро Perl (с версии 5.6.2)
            \item Множество тестов написаны на нём
        \end{itemize}
        \item Простой и понятный API
        \vspace{0.25em}
        \begin{itemize}
            \setlength{\itemsep}{0.5ex}
            \item \texttt{ok(), is(), like(), done\_testing()}
            \item Легко освоить новичкам
        \end{itemize}
        \item Покрывает 80\% потребностей
        \vspace{0.25em}
        \begin{itemize}
            \setlength{\itemsep}{0.5ex}
            \item Базовые проверки
            \item Для сложных сценариев требуются дополнительные модули (\texttt{Test::Deep, Test::MockObject} и т.д.)
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Test2 - революция или эволюция?}
    \begin{itemize}
        \item \textbf{Эволюция в использовании}
        \begin{itemize}
            \item Совместимость: \texttt{Test::Builder} теперь слой над \texttt{Test2}
            \vspace{0.25em}
            \item Привычные \texttt{ok/is/like/subtest} есть в \texttt{Test2::V0}
            \vspace{0.25em}
            \item Улучшения "из коробки": красивый \texttt{Compare} с таблицами, \texttt{Subtest} с буферизацией
        \end{itemize}
        \vspace{0.5em}
        \item \textbf{Революция в архитектуре}
        \begin{itemize}
            \item События вместо глобального состояния
            \vspace{0.25em}
            \item Плагины могут безопасно модифицировать поведение
            \vspace{0.25em}
            \item Легко создавать библиотеки не опасаясь сломать чужой код
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Что такое \texttt{Test2}}
    \begin{itemize}
        \item \texttt{Test2} - это фреймворк для написания тестовых инструментов
        \vspace{0.25em}
        \item Упрощает написание собственных расширений
        \vspace{0.25em}
        \item Предоставляет множество инструментов из коробки \texttt{Test2::Suite}
    \end{itemize}
    \slidefoot{Подробнее: \url{https://perldoc.perl.org/Test2::Suite}}
\end{frame}

\begin{frame}[fragile]{А что будет с \texttt{Test::Builder}?}
    \begin{itemize}
        \item Сейчас \texttt{Test::Builder} - совместимый слой поверх \texttt{Test2}
        \vspace{0.25em}
        \item В таком виде с 10 мая 2016 года
        \vspace{0.25em}
        \item Можно смешивать инструменты из \texttt{Test::Builder} и \texttt{Test2}
        \vspace{0.25em}
        \item Вероятно он так и останется для совместимости со старым кодом
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{\texttt{Прочее}}
    \only<1>{
        \begin{itemize}
            \item \texttt{Test2::Plugin::SRand}
            \begin{itemize}
                \item Делает прогоны стабильными, но позволяет тестировать случайное поведение
                \vspace{0.25em}
                \item Устанавливает seed равным текущей дате: \texttt{srand("YYYYMMDD")}
                \vspace{0.25em}
                \item Seed можно переопределить переменной окружения \texttt{T2\_RAND\_SEED}
                \vspace{0.25em}
                \item В подробном режиме или при падении сообщает использованный seed
                \vspace{0.25em}
                \item Делает "случайное" поведение воспроизводимым
                \vspace{0.25em}
                \item Каждый день свой seed, изменения от "рандома" всё ещё видны
                \vspace{0.25em}
                \item Можно выключить поведение:\\ \texttt{use Test2::V0 -no\_srand => 1;}
            \end{itemize} 
        \end{itemize}
        }
        \only<2>{
            \begin{itemize}
                \item \texttt{Test2::Plugin::UTF8}
                \begin{itemize}
                    \item "UTF8 везде"
                    \vspace{0.25em}
                    \item Импортирует прагму \texttt{utf8} в текущий файл
                    \vspace{0.25em}
                    \item Устанавливает \texttt{STDOUT} в \texttt{utf8}
                    \vspace{0.25em}
                    \item Устанавливает \texttt{STDERR} в \texttt{utf8}
                    \vspace{0.25em}
                    \item Настраивает текущий форматтер на \texttt{UTF8} для всех дескрипторов
                    \vspace{0.25em}
                    \item Применяет это во всех форматтерах
                    \vspace{0.25em}
                    \item Можно выключить поведение:\\ \texttt{use Test2::V0 -no\_utf8 => 1;}
            \end{itemize}
        \end{itemize}
    }
\end{frame}

\begin{frame}[fragile]{\texttt{Резюме}}

            \vspace{0.25em}

            \vspace{0.25em}
            \item Моки

            \item Хелперы: \texttt{warnings}, \texttt{lives}/\texttt{dies} и т.д.
            \vspace{0.25em}
            \item Совместимость: \texttt{Test::More} поверх Test2 без боли
        \end{itemize}
        \vspace{0.5em}
        \item Для разработчиков библиотек: \texttt{Test2::API}
        \begin{itemize}
            \item Интерфейс для написания тестовых инструментов
            \vspace{0.25em}
            \item Без синглтона: изоляция и предсказуемые хуки
            \vspace{0.25em}
            \item Стабильные контракты вместо монкипатчинга
        \end{itemize}
